1. Audit Zero-Trust IAP Connections (Who accessed what VM and when) 
SELECT
  timestamp,
  protopayload_auditlog.authenticationInfo.principalEmail AS researcher,
  resource.labels.instance_id,
  resource.labels.zone,
  protopayload_auditlog.methodName,
  protopayload_auditlog.status.message AS status
FROM `fast_science_audit_logs.cloudaudit_googleapis_com_activity_*`
WHERE protopayload_auditlog.methodName = "AuthorizeUser"
  AND _PARTITIONDATE >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY)
ORDER BY timestamp DESC
LIMIT 100;
2. Detect GKE AI/ML Workload Node Preemptions & Container OOM Errors 
SELECT
  timestamp,
  resource.labels.cluster_name,
  resource.labels.pod_name,
  resource.labels.container_name,
  severity,
  textPayload
FROM `fast_science_audit_logs.stdout_*`
WHERE (textPayload LIKE "%OOMKilled%" OR textPayload LIKE "%Preempted%" OR severity = "ERROR")
  AND _PARTITIONDATE >= DATE_SUB(CURRENT_DATE(), INTERVAL 24 HOUR)
ORDER BY timestamp DESC
LIMIT 50;
